cmake_minimum_required(VERSION 3.20)
project(ShufangNativeDependencies C)
if(NOT DEFINED LIBMOBI_SOURCE)
  message(FATAL_ERROR "LIBMOBI_SOURCE must point to the pinned libmobi checkout")
endif()
set(USE_LIBXML2 OFF CACHE BOOL "" FORCE)
set(USE_ZLIB OFF CACHE BOOL "" FORCE)
set(USE_MINIZ ON CACHE BOOL "" FORCE)
set(USE_ENCRYPTION OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory("${LIBMOBI_SOURCE}" mobi)
if(WIN32)
  target_sources(mobitool PRIVATE "${LIBMOBI_SOURCE}/tools/win32/getopt.c")
endif()
